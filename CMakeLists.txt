cmake_minimum_required(VERSION 3.24)
project(asr_platform VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ASR_ENABLE_QWEN_NATIVE "Build the pinned CPU-only Qwen validation tools" OFF)
option(ASR_BUILD_TESTS "Build model-free tests" ON)
add_library(asr_core INTERFACE)
add_library(asr::core ALIAS asr_core)
target_include_directories(asr_core INTERFACE core/include)
include(cmake/FoundationDependencies.cmake)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(asr_warnings INTERFACE)
target_compile_options(asr_warnings INTERFACE -Wall -Wextra -Wpedantic)
add_library(asr_engines INTERFACE)
target_include_directories(asr_engines INTERFACE engines/interfaces/include)
target_link_libraries(asr_engines INTERFACE asr::core)
add_library(asr_audio audio/src/wav.cpp audio/src/delivery.cpp)
target_include_directories(asr_audio PUBLIC audio/include)
target_link_libraries(asr_audio PUBLIC asr::core PRIVATE asr_warnings)
add_library(asr_mock engines/mock/src/mock_engine.cpp)
target_include_directories(asr_mock PUBLIC engines/mock/include)
target_link_libraries(asr_mock PUBLIC asr_engines PRIVATE asr_warnings)
add_library(asr_scheduler backend/scheduler/src/scheduler.cpp)
target_include_directories(asr_scheduler PUBLIC backend/scheduler/include)
target_link_libraries(asr_scheduler PUBLIC asr::core PRIVATE asr_warnings)
add_library(asr_worker_executor backend/workers/src/executor.cpp)
target_include_directories(asr_worker_executor PUBLIC backend/workers/include)
target_link_libraries(asr_worker_executor PUBLIC asr_engines PRIVATE asr_warnings)
add_library(asr_session_manager backend/sessions/src/session_manager.cpp)
target_include_directories(asr_session_manager PUBLIC backend/sessions/include)
target_link_libraries(asr_session_manager PUBLIC asr_scheduler asr_worker_executor asr_engines PRIVATE asr_warnings)
add_library(asr_config configs/src/config.cpp)
target_include_directories(asr_config PUBLIC configs/include)
target_link_libraries(asr_config PUBLIC nlohmann_json::nlohmann_json PRIVATE yaml-cpp::yaml-cpp asr_warnings)
add_library(asr_storage storage/src/repository.cpp)
target_include_directories(asr_storage PUBLIC storage/include)
target_link_libraries(asr_storage PUBLIC asr::core nlohmann_json::nlohmann_json PRIVATE asr_warnings)
add_library(asr_baseline benchmark/src/baseline.cpp benchmark/src/load.cpp benchmark/src/sweep.cpp)
add_library(asr_observability observability/src/metrics.cpp observability/src/system_sampler.cpp)
target_include_directories(asr_observability PUBLIC observability/include)
target_link_libraries(asr_observability PUBLIC asr::core nlohmann_json::nlohmann_json Threads::Threads PRIVATE asr_warnings)
add_library(asr_websocket backend/transport/src/websocket.cpp backend/transport/src/api_service.cpp)
target_include_directories(asr_websocket PUBLIC backend/transport/include)
target_link_libraries(asr_websocket PUBLIC asr_engines asr_storage asr_observability asr_config
  OpenSSL::Crypto Threads::Threads PRIVATE asr_warnings)
target_include_directories(asr_baseline PUBLIC benchmark/include)
target_link_libraries(asr_baseline PUBLIC asr_audio asr_config asr_engines asr_storage asr_observability asr_session_manager PRIVATE asr_warnings)
add_executable(asr-cli apps/asr_cli/main.cpp)
target_link_libraries(asr-cli PRIVATE asr_baseline asr_mock asr_session_manager asr_websocket Threads::Threads asr_warnings)

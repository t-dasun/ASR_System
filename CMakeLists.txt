cmake_minimum_required(VERSION 3.24)
project(asr_platform VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ASR_ENABLE_QWEN_NATIVE "Build the pinned Qwen CPU runtime and workers" OFF)
option(ASR_BUILD_TESTS "Build model-free C++ tests" ON)

# Libraries and the single user-facing CLI.
include(cmake/CoreLibraries.cmake)
if(ASR_ENABLE_QWEN_NATIVE)
  include(cmake/NativeQwen.cmake)
  include(cmake/RuntimeTargets.cmake)
endif()
if(ASR_BUILD_TESTS)
  enable_testing()
  include(cmake/Tests.cmake)
endif()

cmake_minimum_required(VERSION 3.24)
project(asr_platform VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ASR_ENABLE_QWEN_NATIVE "Build the pinned CPU-only Qwen validation tools" OFF)
option(ASR_BUILD_TESTS "Build model-free tests" ON)
add_library(asr_core INTERFACE)
add_library(asr::core ALIAS asr_core)
target_include_directories(asr_core INTERFACE core/include)
include(cmake/FoundationDependencies.cmake)
find_package(Threads REQUIRED)
add_library(asr_warnings INTERFACE)
target_compile_options(asr_warnings INTERFACE -Wall -Wextra -Wpedantic)
add_library(asr_engines INTERFACE)
target_include_directories(asr_engines INTERFACE engines/interfaces/include)
target_link_libraries(asr_engines INTERFACE asr::core)
add_library(asr_audio audio/src/wav.cpp audio/src/delivery.cpp)
target_include_directories(asr_audio PUBLIC audio/include)
target_link_libraries(asr_audio PUBLIC asr::core PRIVATE asr_warnings)
add_library(asr_mock engines/mock/src/mock_engine.cpp)
target_include_directories(asr_mock PUBLIC engines/mock/include)
target_link_libraries(asr_mock PUBLIC asr_engines PRIVATE asr_warnings)
add_library(asr_config configs/src/config.cpp)
target_include_directories(asr_config PUBLIC configs/include)
target_link_libraries(asr_config PUBLIC nlohmann_json::nlohmann_json PRIVATE yaml-cpp::yaml-cpp asr_warnings)
add_library(asr_storage storage/src/repository.cpp)
target_include_directories(asr_storage PUBLIC storage/include)
target_link_libraries(asr_storage PUBLIC asr::core nlohmann_json::nlohmann_json PRIVATE asr_warnings)
add_library(asr_baseline benchmark/src/baseline.cpp)
add_library(asr_observability observability/src/metrics.cpp observability/src/system_sampler.cpp)
target_include_directories(asr_observability PUBLIC observability/include)
target_link_libraries(asr_observability PUBLIC asr::core nlohmann_json::nlohmann_json Threads::Threads PRIVATE asr_warnings)
target_include_directories(asr_baseline PUBLIC benchmark/include)
target_link_libraries(asr_baseline PUBLIC asr_audio asr_config asr_engines asr_storage asr_observability PRIVATE asr_warnings)
add_executable(asr-cli apps/asr_cli/main.cpp)
target_link_libraries(asr-cli PRIVATE asr_baseline asr_mock Threads::Threads asr_warnings)
if(ASR_ENABLE_QWEN_NATIVE)
  include(cmake/NativeQwen.cmake)
  add_library(asr_native engines/native/src/native_engine.cpp)
  target_include_directories(asr_native PUBLIC engines/native/include PRIVATE engines/native/src)
  target_link_libraries(asr_native PUBLIC asr_engines PRIVATE nlohmann_json::nlohmann_json asr_warnings)
  add_executable(asr-native-worker engines/native/src/worker_main.cpp)
  target_include_directories(asr-native-worker PRIVATE engines/native/src)
  target_link_libraries(asr-native-worker PRIVATE qwen_cpu nlohmann_json::nlohmann_json asr_warnings)
  target_link_libraries(asr-cli PRIVATE asr_native)
  target_compile_definitions(asr-cli PRIVATE ASR_HAS_QWEN_NATIVE=1)
  if(ASR_BUILD_TESTS)
    add_executable(native-worker-stub tests/integration/native_worker_stub.cpp)
    target_include_directories(native-worker-stub PRIVATE engines/native/src)
    target_link_libraries(native-worker-stub PRIVATE nlohmann_json::nlohmann_json asr_warnings)
    add_executable(native-adapter-test tests/integration/native_adapter_test.cpp)
    target_link_libraries(native-adapter-test PRIVATE asr_native asr_warnings)
    add_test(NAME native_adapter_contracts COMMAND native-adapter-test $<TARGET_FILE:native-worker-stub>)
    set_tests_properties(native_adapter_contracts PROPERTIES LABELS "integration;native-contract")
  endif()
  add_executable(asr-native-probe research/native_qwen/probe.cpp)
  target_link_libraries(asr-native-probe PRIVATE asr::core qwen_cpu)
  target_compile_options(asr-native-probe PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(asr-native-lifecycle research/native_qwen/lifecycle.cpp)
  target_link_libraries(asr-native-lifecycle PRIVATE qwen_cpu)
  target_compile_options(asr-native-lifecycle PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ASR_BUILD_TESTS)
  enable_testing()
  add_executable(measurement-test tests/unit/measurement_test.cpp)
  target_link_libraries(measurement-test PRIVATE asr_observability asr_warnings)
  add_test(NAME measurement_contracts COMMAND measurement-test)
  set_tests_properties(measurement_contracts PROPERTIES LABELS "unit;model-free")
  add_executable(chunk-schedule-test tests/unit/chunk_schedule_test.cpp)
  target_link_libraries(chunk-schedule-test PRIVATE asr::core)
  add_test(NAME chunk_schedule COMMAND chunk-schedule-test)
  set_tests_properties(chunk_schedule PROPERTIES LABELS "unit;model-free")
  add_executable(foundation-test tests/unit/foundation_test.cpp)
  target_link_libraries(foundation-test PRIVATE asr_baseline asr_mock Threads::Threads asr_warnings)
  add_test(NAME foundation_contracts COMMAND foundation-test)
  set_tests_properties(foundation_contracts PROPERTIES LABELS "unit;model-free")
  add_executable(audio-test tests/unit/audio_test.cpp)
  target_link_libraries(audio-test PRIVATE asr_audio asr_mock asr_warnings)
  add_test(NAME audio_contracts COMMAND audio-test)
  set_tests_properties(audio_contracts PROPERTIES LABELS "unit;model-free")
  add_executable(audio-realtime-60-test tests/integration/audio_realtime_60.cpp)
  target_link_libraries(audio-realtime-60-test PRIVATE asr_audio asr_warnings)
  add_test(NAME cli_validate COMMAND asr-cli validate --config ${PROJECT_SOURCE_DIR}/configs/mock_baseline.yaml)
  add_test(NAME cli_dry_run COMMAND asr-cli dry-run --config ${PROJECT_SOURCE_DIR}/configs/mock_baseline.yaml --set dataset.language=zh)
  set_tests_properties(cli_validate cli_dry_run PROPERTIES LABELS "integration;model-free")
  add_test(NAME cli_artifacts COMMAND ${CMAKE_COMMAND}
    -DASR_CLI=$<TARGET_FILE:asr-cli> -DTEST_DIR=${CMAKE_BINARY_DIR}/cli-tests
    -DNATIVE_BUILD=${ASR_ENABLE_QWEN_NATIVE}
    -P ${PROJECT_SOURCE_DIR}/tests/integration/cli_contract.cmake)
  set_tests_properties(cli_artifacts PROPERTIES LABELS "integration;model-free")
endif()
